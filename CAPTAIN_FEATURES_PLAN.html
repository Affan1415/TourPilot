🗄️ New Database Tables Required

-- Emergency Contacts CREATE TABLE emergency_contacts ( id UUID PRIMARY KEY, name TEXT NOT NULL, phone TEXT NOT NULL, type VARCHAR(50), -- coast_guard, port_authority, company, hospital priority INTEGER, is_active BOOLEAN DEFAULT true ); -- Incident Reports CREATE TABLE incident_reports ( id UUID PRIMARY KEY, captain_id UUID REFERENCES staff(id), availability_id UUID REFERENCES availabilities(id), incident_type VARCHAR(50), -- medical, equipment, weather, other severity VARCHAR(20), -- low, medium, high, critical description TEXT, location_lat DECIMAL, location_lng DECIMAL, photos JSONB, reported_at TIMESTAMPTZ DEFAULT NOW() ); -- Fuel Logs CREATE TABLE fuel_logs ( id UUID PRIMARY KEY, boat_id UUID REFERENCES boats(id), captain_id UUID REFERENCES staff(id), availability_id UUID REFERENCES availabilities(id), fuel_level_start DECIMAL, fuel_level_end DECIMAL, engine_hours_start DECIMAL, engine_hours_end DECIMAL, notes TEXT, created_at TIMESTAMPTZ DEFAULT NOW() ); -- Maintenance Issues CREATE TABLE maintenance_issues ( id UUID PRIMARY KEY, boat_id UUID REFERENCES boats(id), reported_by UUID REFERENCES staff(id), title TEXT NOT NULL, description TEXT, severity VARCHAR(20), photos JSONB, status VARCHAR(20) DEFAULT 'open', -- open, in_progress, resolved reported_at TIMESTAMPTZ DEFAULT NOW(), resolved_at TIMESTAMPTZ ); -- Trip Logs (GPS Tracking) CREATE TABLE trip_logs ( id UUID PRIMARY KEY, availability_id UUID REFERENCES availabilities(id), captain_id UUID REFERENCES staff(id), started_at TIMESTAMPTZ, ended_at TIMESTAMPTZ, route_data JSONB, -- Array of {lat, lng, timestamp} distance_nm DECIMAL, max_speed_knots DECIMAL, avg_speed_knots DECIMAL ); -- Captain Time Tracking CREATE TABLE time_entries ( id UUID PRIMARY KEY, captain_id UUID REFERENCES staff(id), clock_in TIMESTAMPTZ NOT NULL, clock_out TIMESTAMPTZ, location_in POINT, location_out POINT, breaks JSONB, -- Array of {start, end} total_hours DECIMAL );
